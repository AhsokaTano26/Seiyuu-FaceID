cmake_minimum_required(VERSION 3.20)
project(seiyuu_faceid_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs geometry)
find_package(Threads REQUIRED)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C/C++ distribution")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
find_path(NLOHMANN_INCLUDE_DIR nlohmann/json.hpp HINTS "${JSON_INCLUDE_DIR}" REQUIRED)
add_library(faceid_core SHARED src/faceid.cc src/image_decode.cc)
target_include_directories(faceid_core PUBLIC include)
target_include_directories(faceid_core SYSTEM PRIVATE "${ORT_INCLUDE_DIR}" "${NLOHMANN_INCLUDE_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(faceid_core PRIVATE "${ORT_LIBRARY}" ${OpenCV_LIBS} Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(faceid_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
if(APPLE)
  set(FACEID_INSTALL_RPATH "@loader_path")
else()
  set(FACEID_INSTALL_RPATH "$ORIGIN")
endif()
set_target_properties(faceid_core PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib" INSTALL_RPATH "${FACEID_INSTALL_RPATH}")
include(CTest)
if(BUILD_TESTING)
  add_executable(faceid_c_api_test tests/c_api_test.cc)
  target_link_libraries(faceid_c_api_test PRIVATE faceid_core)
  # Assertions are required even for release builds.
  target_compile_options(faceid_c_api_test PRIVATE -UNDEBUG)
  add_test(NAME faceid_c_api_boundary COMMAND faceid_c_api_test)
  add_executable(faceid_image_decode_test tests/image_decode_test.cc)
  target_include_directories(faceid_image_decode_test PRIVATE src ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(faceid_image_decode_test PRIVATE faceid_core ${OpenCV_LIBS})
  target_compile_options(faceid_image_decode_test PRIVATE -UNDEBUG)
  add_test(NAME faceid_image_orientation COMMAND faceid_image_decode_test)
  add_executable(faceid_denormal_guard_test tests/denormal_guard_test.cc)
  target_include_directories(faceid_denormal_guard_test PRIVATE src)
  target_link_libraries(faceid_denormal_guard_test PRIVATE Threads::Threads)
  target_compile_options(faceid_denormal_guard_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME faceid_denormal_thread_state COMMAND faceid_denormal_guard_test)
endif()
install(TARGETS faceid_core LIBRARY DESTINATION lib)
install(FILES include/faceid.h DESTINATION include)
